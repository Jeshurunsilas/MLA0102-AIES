% Logical inference rules

% Find the type of a planet
planet_type(Planet, Type) :-
    planet(Planet, Type, _, _).

% Find the distance from the Sun
distance_from_sun(Planet, Distance) :-
    planet(Planet, _, Distance, _).

% Find the orbital period
orbital_period(Planet, Days) :-
    planet(Planet, _, _, Days).

% Find all terrestrial planets
terrestrial_planets(Planets) :-
    findall(Planet, planet(Planet, terrestrial, _, _), Planets).

% Find all gas giant planets
gas_giants(Planets) :-
    findall(Planet, planet(Planet, gas_giant, _, _), Planets).

% Find all ice giant planets
ice_giants(Planets) :-
    findall(Planet, planet(Planet, ice_giant, _, _), Planets).
