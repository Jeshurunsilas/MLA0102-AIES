male(john).
male(david).
male(sam).

female(mary).
female(lisa).
female(anna).
female(emma).

parent(john, david).
parent(mary, david).

parent(john, lisa).
parent(mary, lisa).

parent(david, sam).
parent(anna, sam).

parent(david, emma).
parent(anna, emma).

father(X,Y) :-
    male(X),
    parent(X,Y).

mother(X,Y) :-
    female(X),
    parent(X,Y).

brother(X,Y) :-
    male(X),
    parent(P,X),
    parent(P,Y),
    X \= Y.

sister(X,Y) :-
    female(X),
    parent(P,X),
    parent(P,Y),
    X \= Y.

grandparent(X,Y) :-
    parent(X,Z),
    parent(Z,Y).

grandfather(X,Y) :-
    male(X),
    grandparent(X,Y).

grandmother(X,Y) :-
    female(X),
    grandparent(X,Y).

sibling(X,Y) :-
    parent(P,X),
    parent(P,Y),
    X \= Y.
